defmodule Processes.File.FileUploadAgent do
  use GenServer

  def init(_args) do
    {:ok, %{}}
  end

  def start_link(_args) do
    GenServer.start_link(__MODULE__, [], name: __MODULE__)
  end

  def get_uuid_data(uuid) do
    GenServer.call(__MODULE__, {:get, uuid})
  end

  def put_uuid_data(uuid, data) do
    GenServer.cast(__MODULE__, {:put, uuid, data})
  end

  def handle_call({:get, uuid}, _from, state) do
    value = Map.get(state, uuid)
    dbg(value)
    {:reply, value, state}
  end

  def handle_cast({:put, uuid, data}, state) do
    state = Map.put(state, uuid, data)
    dbg(state)
    {:noreply, state}
  end
end
