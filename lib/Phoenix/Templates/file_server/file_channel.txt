defmodule {app_name}Web.FileChannel do
  use {app_name}Web, :channel
  alias Processes.File.FileUploadSupervisor

  @impl true
  def join("file", payload, socket) do
    if authorized?(payload) do
      {:ok, socket}
    else
      {:error, %{reason: "unauthorized"}}
    end
  end

  # Channels can be used in a request/response fashion
  # by sending replies to requests from the client
  @impl true
  def handle_in("ping", payload, socket) do
    {:reply, {:ok, payload}, socket}
  end



  def handle_in("file-header", payload, socket) do
    socket = initFileState(socket)
    dbg(payload)

    %{"filename" => filename, "uid" => uid, "mime_type" => mime_type, "user_id" => user_id} =
      payload

    FileUploadSupervisor.create_file_server(uid, payload)

    packets = []

    dbg(user_id)

    {:noreply,
     socket
     |> assign(:pid_name, uid)
     |> assign(:file_header, payload)
     |> assign(:file_name, filename)
     |> assign(:mime_type, mime_type)
     |> assign(:packets, packets)
     |> assign(:user_id, user_id)}
  end

  #   IPacket {
  #   sequence: number
  #   data: string
  #   uid: string
  #   last_packet: boolean
  # }

  def handle_in("packet", packet, socket) do
    %{"data" => _data, "sequence" => sequence} = packet
    FileUploadSupervisor.accept_packet(packet)

    {:reply, {:ok, %{sequence: sequence}}, socket}
  end

  def handle_in(
        "done",
        %{"uid" => uid} = _payload,
        socket
      ) do
    # TODO: Broadcast Upload of a new video by user or simply return it
    # TODO: Add uuid for the blob being uploaded after recording in the application
    FileUploadSupervisor.done(uid)
    {:noreply, initFileState(socket)}
  end

  defp initFileState(socket) do
    socket
    |> assign(:file_header, nil)
    |> assign(:file_name, "")
    |> assign(:packets, [])
    |> assign(:pid_name, "")
  end

  defp trim_leading(video) do
    video
  end

  # Add authorization logic here as required.
  defp authorized?(_payload) do
    true
  end
end
