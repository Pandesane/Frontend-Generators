defmodule VendorWeb.Api.{api_module} do
  use VendorWeb, :controller
  alias Vendor.{context}.Mongo{api_module}

  def index(conn, _params) do
    {api_module_lc_plural} = Mongo{api_module}.list()

    conn
    |> assign(:{api_module_lc_plural}, {api_module_lc_plural})
    |> render(:index)
  end

  def show(conn, %{"id" => id}) do
    {api_module_lc_singular} = Mongo{api_module}.get!(id)
    dbg({api_module_lc_singular})

    conn
    |> assign(:{api_module_lc_singular}, {api_module_lc_singular})
    |> render(:show)
  end

  def create(conn, params) do
    {api_module_lc_singular} = Mongo{api_module}.create(params)
    render(conn, :create)
  end

  def update(conn, %{"id" => id} = params) do
    Mongo{api_module}.update(id, params)
    render(conn, :update)
  end

  def delete(conn, %{"id" => id} = params) do
    Mongo{api_module}.delete(id)
    render(conn, :delete)
  end
end
