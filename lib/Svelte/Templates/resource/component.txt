<script lang="ts">
	import { page } from '$app/state';
	import type IVideoskit from '$lib/interfaces/IVideoskit';
	import { deleteVideoSkitForm } from '$lib/remote/video_skit.remote';
	import VideoskitEditForm from './VideoskitEditForm.svelte';

	let { videoskit, relation_id }: { videoskit: IVideoskit; relation_id: string | number } =
		$props();

	let dialogElementEdit: HTMLDialogElement;
	let videoskitEditForm: VideoskitEditForm;
</script>

<div class="relative my-8 flex flex-col gap-4 pl-4">
	<!-- subtle vertical accent -->
	<div
		class="absolute top-1 left-0 h-[85%] w-[3px] rounded-full bg-gradient-to-b from-blue-500 to-blue-200 opacity-70"
	></div>

	<div class="flex flex-col gap-4 md:flex-row md:gap-6">
		<!-- VIDEO (LEFT) -->
		<div class="w-full shrink-0 md:w-[260px]">
			<div class="overflow-hidden rounded-xl bg-slate-100 ring-1 ring-slate-200">
				<!-- svelte-ignore a11y_media_has_caption -->
				<video src={videoskit.video_path} controls class="h-[180px] w-full object-cover"></video>
			</div>
		</div>

		<!-- CONTENT (RIGHT) -->
		<div class="flex flex-1 flex-col gap-3">
			<!-- TITLE -->
			<p class="text-sm leading-snug font-semibold tracking-tight text-slate-900">
				{videoskit.title}
			</p>

			<!-- DESCRIPTION -->
			<p class="text-sm leading-relaxed text-slate-600">
				{videoskit.description}
			</p>

			<!-- HELPER TEXT -->
			<p class="text-xs text-slate-400">
				Visual guide to help patients better understand symptoms, progression, and care steps.
			</p>

			<!-- ACTIONS -->
			<div class="mt-auto flex items-center gap-4 pt-2">
				<form {...deleteVideoSkitForm.for(videoskit.id)}>
					<input type="hidden" name="id" value={videoskit.id} />
					<input type="hidden" name="redirect" value={page.url.href} />
					<input type="hidden" name="relation_id" value={relation_id} />

					<button
						onclick={(e) => {
							let btn = e.target as HTMLButtonElement;
							btn.innerText = 'Deleting...';
						}}
						class="flex items-center gap-1 text-xs font-medium text-red-600 transition hover:text-red-700"
					>
						<span class="icon-[hugeicons--delete-02] size-4"></span>
						Delete
					</button>
				</form>

				<button
					onclick={() => {
						dialogElementEdit.showModal();
					}}
					class="flex items-center gap-1 text-xs font-medium text-blue-600 transition hover:text-blue-700"
				>
					<span class="icon-[hugeicons--edit-02] size-4"></span>
					Edit
				</button>
			</div>
		</div>
	</div>
</div>

<!-- Edit Videoskit File -->
<dialog bind:this={dialogElementEdit} class="modal">
	<div class="modal-box max-h-[80vh]">
		<VideoskitEditForm
			bind:this={videoskitEditForm}
			closeModal={() => {
				dialogElementEdit.close();
			}}
			{videoskit}
			{relation_id}
		/>

		<div class="modal-action">
			<button
				onclick={() => {
					dialogElementEdit.close();
					videoskitEditForm.resetForm();
					videoskitEditForm.cleanUpForm();
				}}
				class="btn my-4 btn-primary">Close</button
			>
		</div>
	</div>
</dialog>
