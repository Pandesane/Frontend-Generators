<script lang="ts">
	import FileInputUploader from '$lib/components/FileInputUploader.svelte';
	import type { FileUploadStatus, FormValidation } from '$lib/interfaces/types';
	import { page } from '$app/state';
	import { onMount } from 'svelte';
	import { createNewVideoSkitForm } from '$lib/remote/video_skit.remote';
	import { NewVideoSkitFormSchema } from '$lib/schemas/VideoskitSchema';
	import * as v from 'valibot';
	// Defaults
	let {
		closeModal,
		relation_id
	}: {
		closeModal: () => void;
		relation_id: string | number;
	} = $props();

	let form: HTMLFormElement;
	let submitBtn: HTMLButtonElement;
	let formStatus = $state('Enabled after file is uploaded to server');
	//File Variables
	let fileUploader: FileInputUploader;

	//File Variables
	let accept = 'video/*';
	let uuid: string | undefined = $state();

	onMount(() => {
		uuid = self.crypto.randomUUID();
		// submitBtn.disabled = true;
	});
	function uploadDoneCallBack(status: FileUploadStatus) {
		if (status == 'finished') {
			submitBtn.disabled = false;
			formStatus = 'Finished Uploading File to server. Please fill in the required form fields';
		} else {
			submitBtn.disabled = true;
			formStatus = 'Finished Uploading File to server.';
		}
	}

	export function resetForm() {
		form.reset();
		fileUploader.cleanUpResources();
		uploadDoneCallBack('start');
	}
</script>

<form
	bind:this={form}
	class="mx-auto my-6 flex w-full max-w-200 flex-col gap-5"
	{...createNewVideoSkitForm.enhance(async ({ data, submit }) => {
		await submit();
		if (v.safeParse(NewVideoSkitFormSchema, data).success) {
			closeModal();
			resetForm();
		}
	})}
>
	<!-- HEADER -->
	<div class="flex flex-col items-center space-y-2 text-center">
		<div class="flex items-center justify-center rounded-full bg-blue-100 p-2.5 text-blue-600">
			<span class="icon-[hugeicons--video-01] size-5"></span>
		</div>

		<p class="text-lg font-semibold tracking-tight text-slate-900">Create Video Learning Skit</p>

		<p class="text-xs text-slate-500">
			Upload educational media to visually explain symptoms, causes, or treatment guidance
		</p>
	</div>

	<!-- HIDDEN -->
	<input type="hidden" name="redirect" value={page.url.href} />
	<input type="hidden" name="relation_id" value={relation_id} />

	<!-- TITLE -->
	<div class="flex flex-col gap-1.5">
		<label class="text-xs font-medium text-slate-600">Video Title</label>

		<input
			type="text"
			id="title"
			{...createNewVideoSkitForm.fields.title.as('text')}
			class="rounded-lg border border-slate-300 bg-white px-3 py-2 text-sm
			transition outline-none
			focus:border-blue-500 focus:ring-2 focus:ring-blue-100"
		/>

		{#each createNewVideoSkitForm.fields.title.issues() as issue}
			<p class="text-xs text-red-500">
				{issue.message}
			</p>
		{/each}
	</div>

	<!-- DESCRIPTION -->
	<div class="flex flex-col gap-1.5">
		<label class="text-xs font-medium text-slate-600">Description</label>

		<textarea
			id="description"
			{...createNewVideoSkitForm.fields.description.as('text')}
			class="h-28 resize-none rounded-lg border border-slate-300 bg-white px-3 py-2 text-sm
			transition outline-none
			focus:border-blue-500 focus:ring-2 focus:ring-blue-100"
			placeholder="Explain what this video demonstrates clinically..."
		></textarea>

		{#each createNewVideoSkitForm.fields.description.issues() as issue}
			<p class="text-xs text-red-500">
				{issue.message}
			</p>
		{/each}
	</div>

	<!-- UPLOAD SECTION (VISUALLY EMPHASIZED) -->
	<div class="flex flex-col gap-2 rounded-xl border border-slate-200 bg-slate-50 p-3">
		<div class="flex w-full justify-between">
			<p class="text-xs font-medium text-slate-600">Upload Video File</p>

			<button
			  type="button"
				onclick={() => {
					console.log('Start recording the video');
				}}>Record Video</button
			>
		</div>

		<input value={uuid} type="hidden" name="uuid" />

		<FileInputUploader
			bind:this={fileUploader}
			{accept}
			bind:uuid={uuid!}
			{uploadDoneCallBack}
			label="Select or Drop Video File"
		/>

		<p class="text-xs text-slate-400">
			Supported formats: MP4, WebM — used for clinical demonstration and patient education
		</p>
	</div>

	<!-- CTA -->
	<button
		disabled
		bind:this={submitBtn}
		class="w-full rounded-xl bg-indigo-600 px-4 py-2.5 text-sm font-semibold text-white
		transition-all duration-200
		hover:bg-indigo-700 hover:shadow-sm
		active:scale-[0.98]
		disabled:cursor-not-allowed disabled:bg-slate-300 disabled:text-slate-500 disabled:shadow-none"
	>
		Create Video Skit
	</button>

	<p class="text-center text-xs text-slate-500">
		{formStatus}
	</p>
</form>
