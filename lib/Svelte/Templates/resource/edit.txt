<script lang="ts">
	import type { FileUploadStatus, FormValidation } from '$lib/interfaces/types';
	import FileInputUploader from '$lib/components/FileInputUploader.svelte';
	import { page } from '$app/state';
	import type IVideoskit from '$lib/interfaces/IVideoskit';
	import { onMount } from 'svelte';
	import { editVideoSkitForm } from '$lib/remote/video_skit.remote';
	import { EditVideoSkitFormSchema } from '$lib/schemas/VideoskitSchema';
	import * as v from 'valibot';

	// Defaults
	let {
		videoskit,
		closeModal,
		relation_id
	}: {
		videoskit: IVideoskit;
		closeModal: () => void;
		relation_id: string | number;
	} = $props();

	let formStatus = $state('Enabled after file is uploaded to server');
	let form: HTMLFormElement;
	let submitBtn: HTMLButtonElement;
	let uuid: string | undefined = $state();

	//File Variables
	let fileUploader: FileInputUploader;
	let accept = 'video/*';

	function uploadDoneCallBack(status: FileUploadStatus) {
		if (status == 'finished') {
			formStatus = 'Finished Uploading File to server. Please fill in the required form fields';
			submitBtn.disabled = false;
		} else {
			formStatus = 'Finished Uploading File to server.';
			submitBtn.disabled = true;
		}
	}

	export function resetForm() {
		// form.reset();
		fileUploader.cleanUpResources();
	}

	export function cleanUpForm() {
		fileUploader.cleanUpResources();
	}

	onMount(() => {
		uuid = self.crypto.randomUUID();
		submitBtn.disabled = false;
	});
</script>

<form
	bind:this={form}
	class="mx-auto my-6 flex w-full max-w-200 flex-col gap-5"
	{...editVideoSkitForm.for(videoskit.id).enhance(async ({ data, submit, form }) => {
		await submit();
		if (v.safeParse(EditVideoSkitFormSchema, data).success) {
			closeModal();
			cleanUpForm();
		}
	})}
>
	<!-- HEADER -->
	<div class="flex flex-col items-center space-y-2 text-center">
		<div class="flex items-center justify-center rounded-full bg-blue-100 p-2.5 text-blue-600">
			<span class="size-5 icon-[hugeicons--video-edit-01]"></span>
		</div>

		<p class="text-lg font-semibold tracking-tight text-slate-900">Update Video Learning Skit</p>

		<p class="text-xs text-slate-500">
			Refine educational content, update clinical explanations, or replace media assets
		</p>
	</div>

	<!-- HIDDEN -->
	<input type="hidden" name="id" value={videoskit.id} />
	<input type="hidden" name="redirect" value={page.url.href} />
	<input type="hidden" name="relation_id" value={relation_id} />

	<!-- TITLE -->
	<div class="flex flex-col gap-1.5">
		<label class="text-xs font-medium text-slate-600">Video Title</label>

		<input
			type="text"
			id="title"
			{...editVideoSkitForm.fields.title.as('text')}
			value={videoskit.title}
			class="rounded-lg border border-slate-300 bg-white px-3 py-2 text-sm
			transition outline-none
			focus:border-blue-500 focus:ring-2 focus:ring-blue-100"
		/>

		{#each editVideoSkitForm.fields.title.issues() as issue}
			<p class="text-xs text-red-500">
				{issue.message}
			</p>
		{/each}
	</div>

	<!-- DESCRIPTION -->
	<div class="flex flex-col gap-1.5">
		<label class="text-xs font-medium text-slate-600">Description</label>

		<textarea
			id="description"
			{...editVideoSkitForm.fields.description.as('text')}
			class="h-28 resize-none rounded-lg border border-slate-300 bg-white px-3 py-2 text-sm
			transition outline-none
			focus:border-blue-500 focus:ring-2 focus:ring-blue-100">{videoskit.description}</textarea
		>

		{#each editVideoSkitForm.fields.description.issues() as issue}
			<p class="text-xs text-red-500">
				{issue.message}
			</p>
		{/each}
	</div>

	<!-- MEDIA REPLACEMENT SECTION -->
	<div class="flex flex-col gap-2 rounded-xl border border-slate-200 bg-slate-50 p-3">
		<p class="text-xs font-medium text-slate-600">Replace Video File (Optional)</p>

		<input value={uuid} type="hidden" name="uuid" />

		<FileInputUploader
			bind:this={fileUploader}
			{accept}
			bind:uuid={uuid!}
			{uploadDoneCallBack}
			label="Update Video File"
		/>

		<p class="text-xs text-slate-400">Leave unchanged if current video remains clinically valid</p>
	</div>

	<!-- CTA -->
	<button
		bind:this={submitBtn}
		class="w-full rounded-xl bg-blue-600 px-4 py-2.5 text-sm font-semibold text-white
		transition hover:bg-blue-700 active:scale-[0.98]"
	>
		Save Changes
	</button>

	<p class="text-center text-xs text-slate-500">Update Video Skit</p>
</form>
