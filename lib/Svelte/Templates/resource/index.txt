<script lang="ts">
	import { getVideoSkits } from '$lib/remote/video_skit.remote';
	import VideoskitComponent from './VideoskitComponent.svelte';
	import VideoskitNewForm from './VideoskitNewForm.svelte';
	let dialogElementAdd: HTMLDialogElement;

	let { relation_id }: { relation_id: string | number } = $props();
	let videoskitNewForm: VideoskitNewForm;

	let videoskits = $derived(await getVideoSkits());
</script>

<div class="mx-2 mt-4">
	<div class="mb-5 flex items-start justify-between border-b border-slate-100 pb-3">
		<!-- TITLE BLOCK -->
		<div class="flex flex-col gap-1">
			<p class="text-sm font-semibold tracking-tight text-slate-900">Videoskits</p>

				Visual explanations for understanding and self-management
			</p>
		</div>

		<!-- ACTION -->
		<button
			onclick={() => {
				dialogElementAdd.showModal();
				videoskitNewForm.resetForm();
			}}
			class="flex items-center gap-1 text-xs font-medium text-blue-600 transition hover:text-blue-700"
		>
			<span class="icon-[hugeicons--add-01] size-4"></span>
			Add
		</button>
	</div>

	<div class="mt-4 w-full">
		{#each videoskits as videoskit}
			<VideoskitComponent {relation_id} {videoskit} />
		{/each}
	</div>
</div>

<!-- Add New VideoskitFile -->
<dialog bind:this={dialogElementAdd} class="modal">
	<div class="modal-box max-h-[80vh]">
		<VideoskitNewForm
			bind:this={videoskitNewForm}
			closeModal={() => {
				dialogElementAdd.close();
			}}
			{relation_id}
		/>

		<div class="modal-action">
			<button
				onclick={() => {
					dialogElementAdd.close();
					videoskitNewForm.resetForm();
				}}
				class="btn my-4 btn-primary">Close</button
			>
		</div>
	</div>
